# M1 — Каркас Пылесоса. План реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Один локальный Go-бинарник со встроенным React-фронтом: SQLite с миграциями, задания с курсором и паузой, поток событий, окно программы и три экрана (Источники, Задания, Настройки) — фундамент, на который этапы M2–M6 добавляют исполнителей и экраны.

**Architecture:** `cmd/pylesos` собирает зависимости через конструкторы. Модули `internal/<модуль>/{domain,repository,service}`; `internal/screens` собирает для фронта готовые экраны (тексты, статусы, кнопки), `internal/api` отдаёт их по HTTP на `127.0.0.1` с токеном запуска и шлёт события SSE. Всё длительное — задания (`internal/jobs`): исполнитель (`Worker`) сохраняет курсор, `Runner` ведёт статусы, паузу, продолжение после перезапуска. Фронт только рендерит.

**Tech Stack:** Go 1.26 (`GOTOOLCHAIN=auto`), `modernc.org/sqlite`, `github.com/jmoiron/sqlx`, `log/slog`; React 18, Vite 5, TypeScript 5 (strict), Tailwind 3.4, TanStack Query 5, class-variance-authority, Radix Slot; Node 18.

// Файл: internal/paths/paths.go
// Package paths — где программа хранит данные и выгрузки (спецификация, раздел 1).
package paths

import (
	"path/filepath"
	"strings"
)

// AppDir — имя папки программы в папке данных и в «Документах».
const AppDir = "Пылесос"

// Paths — папки программы.
type Paths struct {
	DataDir    string // база, профиль браузера, временные файлы
	ExportsDir string // выгрузки Excel по умолчанию
}

// Resolve собирает пути из папки настроек ОС (os.UserConfigDir) и папки «Документы».
func Resolve(configDir, documentsDir string) Paths {
	return Paths{
		DataDir:    filepath.Join(configDir, AppDir),
		ExportsDir: filepath.Join(documentsDir, AppDir),
	}
}

// DocumentsDir — папка «Документы» пользователя. На Linux читается XDG_DOCUMENTS_DIR из
// ~/.config/user-dirs.dirs (там она может называться «Документы»), иначе — ~/Documents.
func DocumentsDir(goos, home string, readFile func(string) ([]byte, error)) string {
	fallback := filepath.Join(home, "Documents")
	if goos != "linux" {
		return fallback
	}
	data, err := readFile(filepath.Join(home, ".config", "user-dirs.dirs"))
	if err != nil {
		return fallback
	}
	for _, line := range strings.Split(string(data), "\n") {
		value, ok := strings.CutPrefix(strings.TrimSpace(line), "XDG_DOCUMENTS_DIR=")
		if !ok {
			continue
		}
		dir := filepath.Clean(strings.Replace(strings.Trim(value, `"`), "$HOME", home, 1))
		if dir == filepath.Clean(home) || !filepath.IsAbs(dir) {
			return fallback
		}
		return dir
	}
	return fallback
}
```

// Файл: internal/store/db.go
// Package store открывает базу SQLite (без CGO, П14) и применяет миграции.
package store

import (
	"context"
	"errors"
	"fmt"
	"os"
	"path/filepath"

	"github.com/jmoiron/sqlx"
	_ "modernc.org/sqlite" // регистрирует драйвер "sqlite"
)

const pragmas = "?_pragma=foreign_keys(1)&_pragma=journal_mode(WAL)&_pragma=busy_timeout(5000)"

// Open открывает (или создаёт) базу по пути path и применяет миграции.
func Open(ctx context.Context, path string) (*sqlx.DB, error) {
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return nil, fmt.Errorf("папка базы: %w", err)
	}
	db, err := sqlx.Open("sqlite", path+pragmas)
	if err != nil {
		return nil, fmt.Errorf("открыть базу: %w", err)
	}
	db.SetMaxOpenConns(1) // один писатель: SQLite не любит параллельную запись
	if err := migrate(ctx, db); err != nil {
		return nil, errors.Join(err, db.Close())
	}
	return db, nil
}

// Файл: internal/store/migrate.go
package store

import (
	"context"
	"embed"
	"errors"
	"fmt"
	"io/fs"
	"path"
	"sort"

	"github.com/jmoiron/sqlx"
)

//go:embed migrations/*.sql
var migrationsFS embed.FS

// migrate применяет по порядку миграции, которых ещё нет в schema_migrations.
func migrate(ctx context.Context, db *sqlx.DB) error {
	if _, err := db.ExecContext(ctx, `CREATE TABLE IF NOT EXISTS schema_migrations (version TEXT PRIMARY KEY)`); err != nil {
		return fmt.Errorf("таблица миграций: %w", err)
	}
	names, err := fs.Glob(migrationsFS, "migrations/*.sql")
	if err != nil {
		return fmt.Errorf("список миграций: %w", err)
	}
	sort.Strings(names)
	for _, name := range names {
		if err := applyOne(ctx, db, name); err != nil {
			return err
		}
	}
	return nil
}

func applyOne(ctx context.Context, db *sqlx.DB, name string) error {
	version := path.Base(name)
	var n int
	if err := db.GetContext(ctx, &n, `SELECT COUNT(*) FROM schema_migrations WHERE version = ?`, version); err != nil {
		return fmt.Errorf("проверить миграцию %s: %w", version, err)
	}
	if n > 0 {
		return nil
	}
	body, err := migrationsFS.ReadFile(name)
	if err != nil {
		return fmt.Errorf("прочитать миграцию %s: %w", version, err)
	}
	tx, err := db.BeginTxx(ctx, nil)
	if err != nil {
		return fmt.Errorf("начать миграцию %s: %w", version, err)
	}
	if _, err := tx.ExecContext(ctx, string(body)); err != nil {
		return errors.Join(fmt.Errorf("применить миграцию %s: %w", version, err), tx.Rollback())
	}
	if _, err := tx.ExecContext(ctx, `INSERT INTO schema_migrations (version) VALUES (?)`, version); err != nil {
		return errors.Join(fmt.Errorf("отметить миграцию %s: %w", version, err), tx.Rollback())
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("завершить миграцию %s: %w", version, err)
	}
	return nil
}
```

// Файл: internal/settings/repository/sqlite.go
package repository

import (
	"context"
	"errors"
	"fmt"

	"github.com/jmoiron/sqlx"
)

// SQLite — настройки в таблице settings.
type SQLite struct{ db *sqlx.DB }

func NewSQLite(db *sqlx.DB) *SQLite { return &SQLite{db: db} }

type row struct {
	Key   string `db:"key"`
	Value string `db:"value"`
}

func (r *SQLite) All(ctx context.Context) (map[string]string, error) {
	var rows []row
	if err := r.db.SelectContext(ctx, &rows, `SELECT key, value FROM settings`); err != nil {
		return nil, fmt.Errorf("прочитать настройки: %w", err)
	}
	out := make(map[string]string, len(rows))
	for _, x := range rows {
		out[x.Key] = x.Value
	}
	return out, nil
}

func (r *SQLite) Put(ctx context.Context, values map[string]string) error {
	tx, err := r.db.BeginTxx(ctx, nil)
	if err != nil {
		return fmt.Errorf("начать запись настроек: %w", err)
	}
	for k, v := range values {
		if _, err := tx.ExecContext(ctx, `INSERT INTO settings (key, value) VALUES (?, ?)
			ON CONFLICT (key) DO UPDATE SET value = excluded.value`, k, v); err != nil {
			return errors.Join(fmt.Errorf("записать настройку %s: %w", k, err), tx.Rollback())
		}
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("сохранить настройки: %w", err)
	}
	return nil
}

package service

import (
	"context"
	"fmt"
	"strconv"

	"pylesos/internal/settings/domain"
	"pylesos/internal/settings/repository"
)

type Service struct {
	repo     repository.Repository
	defaults domain.Settings
}

func NewService(repo repository.Repository, defaults domain.Settings) *Service {
	return &Service{repo: repo, defaults: defaults}
}

// Get — сохранённые настройки поверх значений по умолчанию; испорченное значение заменяется умолчанием.
func (s *Service) Get(ctx context.Context) (domain.Settings, error) {
	stored, err := s.repo.All(ctx)
	if err != nil {
		return domain.Settings{}, fmt.Errorf("настройки: %w", err)
	}
	out := s.defaults
	if v := stored[domain.KeyExportsDir]; v != "" {
		out.ExportsDir = v
	}
	if n, convErr := strconv.Atoi(stored[domain.KeyPhonePauseSec]); convErr == nil && inPauseRange(n) {
		out.PhonePauseSec = n
	}
	return out, nil
}

// Save проверяет ввод пользователя и сохраняет его. Ошибки проверки возвращаются по ключам полей
// готовым русским текстом; при них ничего не сохраняется, а первый результат — текущие настройки.
func (s *Service) Save(ctx context.Context, input map[string]string) (domain.Settings, map[string]string, error) {
	values, problems := validate(input)
	if len(problems) > 0 {
		current, err := s.Get(ctx)
		return current, problems, err
	}
	if err := s.repo.Put(ctx, values); err != nil {
		return domain.Settings{}, nil, fmt.Errorf("сохранить настройки: %w", err)
	}
	saved, err := s.Get(ctx)
	return saved, nil, err
}

import (
	"fmt"
	"path/filepath"

	"pylesos/internal/settings/domain"
	"pylesos/internal/texts"
)

// validate возвращает очищенные значения для записи и ошибки по ключам полей.
func validate(input map[string]string) (map[string]string, map[string]string) {
	values := map[string]string{}
	problems := map[string]string{}
	dir := strings.TrimSpace(input[domain.KeyExportsDir])
	switch {
	case dir == "":
		problems[domain.KeyExportsDir] = texts.ErrExportsDirEmpty
	case !filepath.IsAbs(dir):
		problems[domain.KeyExportsDir] = texts.ErrExportsDirRelative
	default:
		values[domain.KeyExportsDir] = filepath.Clean(dir)
	}
	pause, err := strconv.Atoi(strings.TrimSpace(input[domain.KeyPhonePauseSec]))
	switch {
	case err != nil:
		problems[domain.KeyPhonePauseSec] = texts.ErrPauseNotNumber
	case !inPauseRange(pause):
		problems[domain.KeyPhonePauseSec] = fmt.Sprintf(texts.ErrPauseRange, domain.MinPhonePauseSec, domain.MaxPhonePauseSec)
	default:
		values[domain.KeyPhonePauseSec] = strconv.Itoa(pause)
	}
	return values, problems
}

func inPauseRange(n int) bool {
	return n >= domain.MinPhonePauseSec && n <= domain.MaxPhonePauseSec
}

// Файл: internal/events/bus.go
// Package events — оповещение открытых окон об изменениях (поток SSE).
// События — только сигнал «перечитай экран», поэтому потерять лишнее не страшно.
package events

import "sync"

// Типы событий.
const JobsChanged = "jobs_changed"

type Event struct {
	Type string `json:"type"`
}

const subscriberBuffer = 16

type Bus struct {
	mu   sync.Mutex
	subs map[chan Event]struct{}
}

func NewBus() *Bus { return &Bus{subs: map[chan Event]struct{}{}} }

// Subscribe возвращает канал событий и функцию отписки (её можно вызывать повторно).
func (b *Bus) Subscribe() (<-chan Event, func()) {
	ch := make(chan Event, subscriberBuffer)
	b.mu.Lock()
	b.subs[ch] = struct{}{}
	b.mu.Unlock()
	var once sync.Once
	return ch, func() {
		once.Do(func() {
			b.mu.Lock()
			delete(b.subs, ch)
			b.mu.Unlock()
			close(ch)
		})
	}
}

// Publish рассылает событие; подписчику с полным буфером событие не достаётся — программа не ждёт.
func (b *Bus) Publish(e Event) {
	b.mu.Lock()
	defer b.mu.Unlock()
	for ch := range b.subs {
		select {
		case ch <- e:
		default:
		}
	}
}

	StatusFailed      Status = "failed"
	StatusCancelled   Status = "cancelled"

	ActionPause  Action = "pause"
	ActionResume Action = "resume"
	ActionCancel Action = "cancel"
)

func (s Status) Actions() []Action {

	case StatusRunning:
		return []Action{ActionPause, ActionCancel}
	case StatusPaused, StatusWaitingUser, StatusFailed:
		return []Action{ActionResume, ActionCancel}
	case StatusQueued:
		return []Action{ActionCancel}
	default:
		return []Action{}
	}
}

import (
	"context"

	sqlite3 "modernc.org/sqlite/lib"

const columns = `id, source_id, kind, params, status, cursor, progress_text, reason, created_at, updated_at, finished_at`

type SQLite struct{ db *sqlx.DB }

func NewSQLite(db *sqlx.DB) *SQLite { return &SQLite{db: db} }

func (r *SQLite) Create(ctx context.Context, j domain.Job) (int64, error) {
	res, err := r.db.NamedExecContext(ctx, `INSERT INTO jobs (source_id, kind, params, status, cursor, created_at, updated_at)
		VALUES (:source_id, :kind, :params, :status, :cursor, :created_at, :updated_at)`, j)
	if err != nil {
		return 0, busyOr(fmt.Errorf("создать задание: %w", err))
	}
	id, err := res.LastInsertId()
	if err != nil {
		return 0, fmt.Errorf("id задания: %w", err)
	}
	return id, nil
}

func (r *SQLite) Get(ctx context.Context, id int64) (domain.Job, error) {
	var j domain.Job
	err := r.db.GetContext(ctx, &j, `SELECT `+columns+` FROM jobs WHERE id = ?`, id)
	if errors.Is(err, sql.ErrNoRows) {
		return j, domain.ErrNotFound
	}
	if err != nil {
		return j, fmt.Errorf("прочитать задание %d: %w", id, err)
	}
	return j, nil
}

func (r *SQLite) List(ctx context.Context) ([]domain.ListItem, error) {
	items := []domain.ListItem{}
	err := r.db.SelectContext(ctx, &items, `SELECT j.id, j.source_id, j.kind, j.params, j.status, j.cursor,
		j.progress_text, j.reason, j.created_at, j.updated_at, j.finished_at, s.title AS source_title
		FROM jobs j JOIN sources s ON s.id = j.source_id ORDER BY j.id DESC`)
	if err != nil {
		return nil, fmt.Errorf("список заданий: %w", err)
	}
	return items, nil
}

func (r *SQLite) ListByStatus(ctx context.Context, statuses ...domain.Status) ([]domain.Job, error) {
	query, args, err := sqlx.In(`SELECT `+columns+` FROM jobs WHERE status IN (?) ORDER BY id`, statuses)
	if err != nil {
		return nil, fmt.Errorf("запрос заданий по статусу: %w", err)
	}
	jobs := []domain.Job{}
	if err := r.db.SelectContext(ctx, &jobs, r.db.Rebind(query), args...); err != nil {
		return nil, fmt.Errorf("задания по статусу: %w", err)
	}
	return jobs, nil
}

func (r *SQLite) UpdateState(ctx context.Context, id int64, u domain.StateUpdate) error {
	var finished sql.NullInt64
	if u.Finished {
		finished = sql.NullInt64{Int64: u.At, Valid: true}
	}
	res, err := r.db.ExecContext(ctx, `UPDATE jobs SET status = ?, reason = ?, updated_at = ?, finished_at = ? WHERE id = ?`,
		u.Status, u.Reason, u.At, finished, id)
	if err != nil {
		return busyOr(fmt.Errorf("обновить задание %d: %w", id, err))
	}
	return affected(res, id)
}

func (r *SQLite) SaveCursor(ctx context.Context, id int64, cursor, progressText string, at int64) error {
	res, err := r.db.ExecContext(ctx, `UPDATE jobs SET cursor = ?, progress_text = ?, updated_at = ? WHERE id = ?`,
		cursor, progressText, at, id)
	if err != nil {
		return fmt.Errorf("сохранить курсор задания %d: %w", id, err)
	}
	return affected(res, id)
}

func affected(res sql.Result, id int64) error {
	n, err := res.RowsAffected()
	if err != nil {
		return fmt.Errorf("задание %d: %w", id, err)
	}
	if n == 0 {
		return domain.ErrNotFound
	}
	return nil
}

// busyOr превращает нарушение индекса jobs_one_active_per_source в domain.ErrSourceBusy.
func busyOr(err error) error {
	var se *sqlite.Error
	if errors.As(err, &se) && se.Code() == sqlite3.SQLITE_CONSTRAINT_UNIQUE {
		return fmt.Errorf("%w: %w", domain.ErrSourceBusy, err)
	}
	return err
}

// Worker — исполнитель заданий одного вида (история Telegram, поиск OLX, разбор… — с M2).
type Worker interface {
	Kind() string  // вид задания, хранится в jobs.kind
	Title() string // название вида для экранов, например «История Telegram»
	// Run выполняет задание с места job.Cursor и сохраняет курсор через p.Save после каждой порции.
	// Возвращает nil — готово; *WaitUserError — нужны действия пользователя; ctx.Err() — пауза, отмена
	// или выход; другую ошибку — сбой (её текст увидит пользователь, поэтому по-русски).
	Run(ctx context.Context, job domain.Job, p Progress) error
}

// Progress — сохранение курсора и строки прогресса выполняющегося задания.
type Progress interface {
	Save(ctx context.Context, cursor, progressText string) error
}

// WaitUserError — задание ждёт пользователя (войти, пройти капчу…). Reason — готовый текст для экрана.
type WaitUserError struct{ Reason string }

func (e *WaitUserError) Error() string { return e.Reason }

// Publisher — куда сообщать об изменениях заданий.
type Publisher interface{ Publish(events.Event) }

var (
	ErrUnknownKind = errors.New("неизвестный вид задания")
	ErrNotRunning  = errors.New("задание сейчас не выполняется")
	ErrNotAllowed  = errors.New("действие недоступно в этом статусе")

// Файл: internal/jobs/service/runner.go
package service

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"slices"
	"sync"
	"time"

	"pylesos/internal/events"
	"pylesos/internal/jobs/domain"
	"pylesos/internal/jobs/repository"
	"pylesos/internal/texts"
)

type runningJob struct {
	cancel context.CancelCauseFunc
	done   chan struct{}
}

type Runner struct {
	base    context.Context // отменяется при выходе из программы
	repo    repository.Repository
	bus     Publisher
	log     *slog.Logger
	now     func() time.Time
	workers map[string]Worker

	mu      sync.Mutex
	running map[int64]runningJob
	wg      sync.WaitGroup
}

func NewRunner(base context.Context, repo repository.Repository, bus Publisher, log *slog.Logger, now func() time.Time, workers ...Worker) *Runner {
	byKind := make(map[string]Worker, len(workers))
	for _, w := range workers {
		byKind[w.Kind()] = w
	}
	return &Runner{base: base, repo: repo, bus: bus, log: log, now: now, workers: byKind, running: map[int64]runningJob{}}
}

// KindTitle — название вида задания для экранов; неизвестный вид показывается как есть.
func (r *Runner) KindTitle(kind string) string {
	if w, ok := r.workers[kind]; ok {
		return w.Title()
	}
	return kind
}

// Start создаёт задание и запускает его.
func (r *Runner) Start(ctx context.Context, sourceID int64, kind, params string) (int64, error) {
	if _, ok := r.workers[kind]; !ok {
		return 0, fmt.Errorf("%w: %s", ErrUnknownKind, kind)
	}
	at := r.now().Unix()
	id, err := r.repo.Create(ctx, domain.Job{SourceID: sourceID, Kind: kind, Params: params,
		Status: domain.StatusQueued, Cursor: "{}", CreatedAt: at, UpdatedAt: at})
	if err != nil {
		return 0, err
	}
	r.changed()
	job, err := r.repo.Get(ctx, id)
	if err != nil {
		return 0, err
	}
	return id, r.launch(ctx, job)
}

// Resume продолжает приостановленное, ждущее пользователя или упавшее задание с сохранённого курсора.
func (r *Runner) Resume(ctx context.Context, id int64) error {
	job, err := r.repo.Get(ctx, id)
	if err != nil {
		return err
	}
	if !slices.Contains(job.Status.Actions(), domain.ActionResume) {
		return ErrNotAllowed
	}
	return r.launch(ctx, job)
}

// Pause останавливает выполняющееся задание и ждёт, пока исполнитель остановится.
func (r *Runner) Pause(ctx context.Context, id int64) error { return r.stop(ctx, id, errPaused) }

// Cancel отменяет задание: выполняющееся — останавливает, остальные (пауза, ждёт, ошибка, очередь) —
// помечает отменёнными. После stop статус в базе уже итоговый: finish пишет его до закрытия done.
func (r *Runner) Cancel(ctx context.Context, id int64) error {
	if err := r.stop(ctx, id, errCancelled); err != nil && !errors.Is(err, ErrNotRunning) {
		return err
	}
	job, err := r.repo.Get(ctx, id)
	if err != nil {
		return err
	}
	if job.Status == domain.StatusCancelled {
		return nil
	}
	if !slices.Contains(job.Status.Actions(), domain.ActionCancel) {
		return ErrNotAllowed
	}
	return r.setState(ctx, id, domain.StatusCancelled, "", true)
}

// ResumeInterrupted при запуске программы продолжает задания, прерванные выходом или сбоем.
func (r *Runner) ResumeInterrupted(ctx context.Context) error {
	jobs, err := r.repo.ListByStatus(ctx, domain.StatusQueued, domain.StatusRunning)
	if err != nil {
		return err
	}
	for _, job := range jobs {
		if err := r.launch(ctx, job); err != nil {
			r.log.Error("продолжить задание", "job", job.ID, "err", err)
		}
	}
	return nil
}

// Wait ждёт остановки всех исполнителей (после отмены base при выходе из программы).
func (r *Runner) Wait() { r.wg.Wait() }

func (r *Runner) stop(ctx context.Context, id int64, cause error) error {
	r.mu.Lock()
	rj, ok := r.running[id]
	r.mu.Unlock()
	if !ok {
		return ErrNotRunning
	}
	rj.cancel(cause)
	select {
	case <-rj.done:
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

func (r *Runner) setState(ctx context.Context, id int64, s domain.Status, reason string, finished bool) error {
	u := domain.StateUpdate{Status: s, Reason: reason, At: r.now().Unix(), Finished: finished}
	if err := r.repo.UpdateState(ctx, id, u); err != nil {
		return err
	}
	r.changed()
	return nil
}

func (r *Runner) changed() { r.bus.Publish(events.Event{Type: events.JobsChanged}) }

// unknownKind помечает задание неизвестного вида упавшим.
func (r *Runner) unknownKind(ctx context.Context, job domain.Job) error {
	return errors.Join(fmt.Errorf("%w: %s", ErrUnknownKind, job.Kind),
		r.setState(ctx, job.ID, domain.StatusFailed, texts.JobUnknownKind, true))
}

// Файл: internal/jobs/service/execute.go
package service

import (
	"context"
	"errors"

	"pylesos/internal/jobs/domain"
	"pylesos/internal/texts"
)

// launch регистрирует задание как выполняющееся, переводит в «идёт» и запускает исполнителя.
// Регистрация до смены статуса: Pause/Cancel в этот момент уже видят задание, а два одновременных
// Resume не запустят его дважды.
func (r *Runner) launch(ctx context.Context, job domain.Job) error {
	w, ok := r.workers[job.Kind]
	if !ok {
		return r.unknownKind(ctx, job)
	}
	jobCtx, cancel := context.WithCancelCause(r.base)
	rj := runningJob{cancel: cancel, done: make(chan struct{})}
	if !r.register(job.ID, rj) {
		cancel(nil)
		return ErrNotAllowed
	}
	if err := r.setState(ctx, job.ID, domain.StatusRunning, "", false); err != nil {
		r.unregister(job.ID)
		cancel(nil)
		close(rj.done)
		return err
	}
	r.wg.Add(1)
	go r.execute(jobCtx, w, job, rj)
	return nil
}

func (r *Runner) register(id int64, rj runningJob) bool {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, busy := r.running[id]; busy {
		return false
	}
	r.running[id] = rj
	return true
}

func (r *Runner) unregister(id int64) {
	r.mu.Lock()
	delete(r.running, id)
	r.mu.Unlock()
}

// execute: итог записывается до закрытия done, поэтому после stop статус в базе уже итоговый.
func (r *Runner) execute(ctx context.Context, w Worker, job domain.Job, rj runningJob) {
	defer r.wg.Done()
	defer close(rj.done)
	defer r.unregister(job.ID)
	err := r.safeRun(ctx, w, job)
	r.finish(ctx, job.ID, err)
	rj.cancel(nil)
}

// safeRun запускает исполнителя; паника превращается в ошибку задания, а не падение программы.
func (r *Runner) safeRun(ctx context.Context, w Worker, job domain.Job) (err error) {
	defer func() {
		if p := recover(); p != nil {
			r.log.Error("сбой исполнителя", "job", job.ID, "panic", p)
			err = errors.New(texts.JobCrashed)
		}
	}()
	return w.Run(ctx, job, progress{runner: r, id: job.ID})
}

// finish записывает итог. При выходе из программы статус не трогается — задание продолжится при запуске.
func (r *Runner) finish(ctx context.Context, id int64, err error) {
	status, reason, finished, ok := outcome(ctx, r.base, err)
	if !ok {
		return
	}
	if err := r.setState(context.WithoutCancel(ctx), id, status, reason, finished); err != nil {
		r.log.Error("сохранить итог задания", "job", id, "err", err)
	}
}

// outcome переводит результат исполнителя в статус: (статус, причина, завершено, записывать ли).
func outcome(ctx, base context.Context, err error) (domain.Status, string, bool, bool) {
	var wait *WaitUserError
	cause := context.Cause(ctx)
	switch {
	case err == nil:
		return domain.StatusDone, "", true, true
	case errors.As(err, &wait):
		return domain.StatusWaitingUser, wait.Reason, false, true
	case errors.Is(cause, errPaused):
		return domain.StatusPaused, "", false, true
	case errors.Is(cause, errCancelled):
		return domain.StatusCancelled, "", true, true
	case base.Err() != nil:
		return "", "", false, false
	default:
		return domain.StatusFailed, err.Error(), true, true
	}
}

type progress struct {
	runner *Runner
	id     int64
}

func (p progress) Save(ctx context.Context, cursor, progressText string) error {
	if err := p.runner.repo.SaveCursor(ctx, p.id, cursor, progressText, p.runner.now().Unix()); err != nil {
		return err
	}
	p.runner.changed()
	return nil
}

	token := d.Token

		buf := make([]byte, 32)
		if _, err := rand.Read(buf); err != nil {
			return nil, fmt.Errorf("токен запуска: %w", err)
		}
		token = base64.RawURLEncoding.EncodeToString(buf)
	}
	return &Server{deps: d, token: token, quit: make(chan struct{}), show: make(chan struct{}, 1)}, nil

// Файл: internal/api/guard.go
package api

import (
	"crypto/subtle"
	"net"
	"net/http"
	"strings"
)

// guard: Host только локальный (защита от DNS rebinding); для /api/* — токен и JSON в POST.
// Образец — doc-agent, internal/ui/server.go.
func (s *Server) guard(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		_, port, err := net.SplitHostPort(s.addr)
		if err != nil || (r.Host != s.addr && r.Host != "localhost:"+port) {
			http.Error(w, "forbidden", http.StatusForbidden)
			return
		}
		if strings.HasPrefix(r.URL.Path, "/api/") {
			token := r.Header.Get(TokenHeader)
			if token == "" {
				token = r.URL.Query().Get("t")
			}
			if subtle.ConstantTimeCompare([]byte(token), []byte(s.token)) != 1 {
				http.Error(w, "forbidden", http.StatusForbidden)
				return
			}
			if r.Method == http.MethodPost && !strings.HasPrefix(r.Header.Get("Content-Type"), "application/json") {
				http.Error(w, "unsupported media type", http.StatusUnsupportedMediaType)
				return
			}
		}
		next.ServeHTTP(w, r)
	})
}

	if err != nil {
		s.writeError(w, http.StatusBadRequest, texts.ErrBadRequest)
		return
	}
	var act func() error
	switch jobsdomain.Action(r.PathValue("action")) {
	case jobsdomain.ActionPause:
		act = func() error { return s.deps.Jobs.Pause(r.Context(), id) }
	case jobsdomain.ActionResume:
		act = func() error { return s.deps.Jobs.Resume(r.Context(), id) }
	case jobsdomain.ActionCancel:
		act = func() error { return s.deps.Jobs.Cancel(r.Context(), id) }
	default:
		s.writeError(w, http.StatusBadRequest, texts.ErrBadRequest)
		return
	}

		s.writeActionError(w, id, err)
		return

// handleEvents — поток событий: окно перечитывает экраны по сигналу.
func (s *Server) handleEvents(w http.ResponseWriter, r *http.Request) {
	flusher, ok := w.(http.Flusher)
	if !ok {
		http.Error(w, "streaming unsupported", http.StatusInternalServerError)
		return
	}
	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	ch, unsubscribe := s.deps.Bus.Subscribe()
	defer unsubscribe()
	if _, err := fmt.Fprint(w, ": подключено\n\n"); err != nil {
		return
	}
	flusher.Flush()
	for {
		select {
		case <-r.Context().Done():
			return
		case e, open := <-ch:
			if !open {
				return
			}
			data, err := json.Marshal(e)
			if err != nil {
				s.deps.Log.Error("событие в JSON", "err", err)
				continue
			}
			if _, err := fmt.Fprintf(w, "data: %s\n\n", data); err != nil {
				return
			}
			flusher.Flush()
		}
	}
}

package window

// MinWindowLife — окно, закрывшееся быстрее, считается не открывшимся (браузер не запустился).
const MinWindowLife = 3 * time.Second

func LivedLongEnough(d time.Duration) bool { return d >= MinWindowLife }

// AppBrowser — путь к браузеру на Chromium, умеющему окно приложения (--app); "" — такого нет.
func AppBrowser(goos string, lookPath func(string) (string, error), exists func(string) bool, getenv func(string) string) string {
	var candidates []string
	switch goos {
	case "windows":
		for _, base := range []string{getenv("ProgramFiles(x86)"), getenv("ProgramFiles"), getenv("LocalAppData")} {
			if base == "" {
				continue
			}
			candidates = append(candidates,
				filepath.Join(base, "Microsoft", "Edge", "Application", "msedge.exe"),
				filepath.Join(base, "Google", "Chrome", "Application", "chrome.exe"))
		}
	case "darwin":
		candidates = []string{
			"/Applications/Google Chrome.app/Contents/MacOS/Google Chrome",
			"/Applications/Microsoft Edge.app/Contents/MacOS/Microsoft Edge",
			"/Applications/Chromium.app/Contents/MacOS/Chromium",
			"/Applications/Brave Browser.app/Contents/MacOS/Brave Browser",
		}
	default:
		for _, name := range []string{"google-chrome", "google-chrome-stable", "chromium", "chromium-browser", "microsoft-edge", "microsoft-edge-stable", "brave-browser"} {
			if p, err := lookPath(name); err == nil {
				return p
			}
		}
		return ""
	}
	for _, p := range candidates {
		if exists(p) {
			return p
		}
	}
	return ""
}

// Opener открывает окно программы и файлы. Всё, что зависит от ОС, — поля, чтобы подменять в тестах.
type Opener struct {
	GOOS       string
	LookPath   func(string) (string, error)
	Exists     func(string) bool
	Getenv     func(string) string
	Start      func(name string, args ...string) (wait func() error, err error)
	ProfileDir string // профиль Chromium окна программы
	Log        *slog.Logger
}

// System — Opener для текущей ОС.
func System(profileDir string, log *slog.Logger) Opener {
	return Opener{
		GOOS:     runtime.GOOS,
		LookPath: exec.LookPath,
		Exists: func(p string) bool {
			_, err := os.Stat(p)
			return err == nil
		},
		Getenv:     os.Getenv,
		Start:      startProcess,
		ProfileDir: profileDir,
		Log:        log,
	}
}

func startProcess(name string, args ...string) (func() error, error) {
	cmd := exec.Command(name, args...)
	if err := cmd.Start(); err != nil {
		return nil, err
	}
	return cmd.Wait, nil
}

// OpenApp показывает окно программы; канал закрывается, когда пользователь закрыл окно.
func (o Opener) OpenApp(link string) (<-chan struct{}, error) {
	if o.GOOS == "linux" && o.Getenv("DISPLAY") == "" && o.Getenv("WAYLAND_DISPLAY") == "" {
		return nil, ErrNoDisplay
	}
	browser := AppBrowser(o.GOOS, o.LookPath, o.Exists, o.Getenv)
	if browser == "" {
		return nil, ErrNoAppBrowser
	}
	wait, err := o.Start(browser, WindowArgs(link, o.ProfileDir)...)
	if err != nil {
		return nil, fmt.Errorf("запустить окно: %w", err)
	}
	done := make(chan struct{})
	go func() {
		defer close(done)
		if err := wait(); err != nil {
			o.Log.Info("окно программы закрыто", "err", err)
		}
	}()
	return done, nil
}

package window

// OpenCommand выбирает программу для открытия target. Папки на Windows — через explorer.exe
// (rundll32 url.dll ломает не-ASCII пути). На Linux без экрана (SSH) — сразу ошибка.
func OpenCommand(goos, target string, dir bool, getenv func(string) string) (string, []string, error) {
	switch goos {
	case "windows":
		if dir {
			return "explorer.exe", []string{target}, nil
		}
		return "rundll32", []string{"url.dll,FileProtocolHandler", target}, nil
	case "darwin":
		return "open", []string{target}, nil
	default:
		if getenv("DISPLAY") == "" && getenv("WAYLAND_DISPLAY") == "" {
			return "", nil, ErrNoDisplay
		}
		return "xdg-open", []string{target}, nil
	}
}

// OpenInOS открывает адрес в браузере по умолчанию, файл — в его программе, папку — в файловом менеджере.
func (o Opener) OpenInOS(target string, dir bool) error {
	name, args, err := OpenCommand(o.GOOS, target, dir, o.Getenv)
	if err != nil {
		return err
	}
	wait, err := o.Start(name, args...)
	if err != nil {
		return err
	}
	go func() {
		if err := wait(); err != nil {
			o.Log.Warn("открыть в ОС", "target", target, "err", err)
		}
	}()
	return nil
}

// Write запоминает адрес запущенной программы (в адресе — токен, поэтому права 0600).
func Write(path, link string) error {
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return fmt.Errorf("папка файла экземпляра: %w", err)
	}
	data, err := json.Marshal(info{URL: link})
	if err != nil {
		return fmt.Errorf("файл экземпляра: %w", err)
	}
	return os.WriteFile(path, data, 0o600)
}

// ShowRunning просит уже запущенную программу показать окно; false — такой нет.
func ShowRunning(ctx context.Context, client *http.Client, path string) bool {
	data, err := os.ReadFile(path)
	if err != nil {
		return false
	}
	var in info
	if json.Unmarshal(data, &in) != nil {
		return false
	}
	u, err := url.Parse(in.URL)
	if err != nil || u.Query().Get("t") == "" {
		return false
	}
	ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
	defer cancel()
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, "http://"+u.Host+"/api/show", strings.NewReader("{}"))
	if err != nil {
		return false
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set(api.TokenHeader, u.Query().Get("t"))
	resp, err := client.Do(req)
	if err != nil {
		return false
	}
	return resp.Body.Close() == nil && resp.StatusCode == http.StatusOK

// Транспорт к бэкенду: только запросы и ошибки, никакой логики (П3).
const token = new URLSearchParams(window.location.search).get('t') ?? ''

export class ApiError extends Error {}

async function parse<T>(res: Response): Promise<T> {
  const body: unknown = await res.json().catch(() => null)
  if (!res.ok) {
    const text =
      typeof body === 'object' && body !== null && 'error' in body
        ? String((body as { error: unknown }).error)
        : res.statusText
    throw new ApiError(text)
  }
  return body as T
}

export async function getJSON<T>(path: string): Promise<T> {
  return parse<T>(await fetch(path, { headers: { 'X-Pylesos-Token': token } }))
}

export async function postJSON<T>(path: string, body: unknown): Promise<T> {
  return parse<T>(
    await fetch(path, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-Pylesos-Token': token },
      body: JSON.stringify(body),
    }),
  )
}

export function eventsURL(): string {
  return `/api/events?t=${encodeURIComponent(token)}`
}

package main

import (
	"context"
	"fmt"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"path/filepath"
	"runtime"

func main() {
	log := logger.New(os.Stderr, slog.LevelInfo)
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	err := run(ctx, log)
	stop()
	if err != nil {
		log.Error("программа остановлена с ошибкой", "err", err)
		os.Exit(1)
	}
}

func run(ctx context.Context, log *slog.Logger) error {
	p, err := resolvePaths()
	if err != nil {
		return err
	}
	if err := os.MkdirAll(p.DataDir, 0o700); err != nil {
		return fmt.Errorf("папка данных: %w", err)
	}
	instanceFile := filepath.Join(p.DataDir, "instance.json")
	if instance.ShowRunning(ctx, http.DefaultClient, instanceFile) {
		log.Info("программа уже запущена — показываю её окно")
		return nil
	}
	db, err := store.Open(ctx, filepath.Join(p.DataDir, "pylesos.db"))
	if err != nil {
		return err
	}
	defer closeLogged(log, "базу", db.Close)
	appCtx, cancelApp := context.WithCancel(ctx)
	defer cancelApp()
	a, err := wire(appCtx, db, p, log)
	if err != nil {
		return err
	}
	return serve(ctx, cancelApp, a, p, instanceFile, log)
}

func resolvePaths() (paths.Paths, error) {
	cfg, err := os.UserConfigDir()
	if err != nil {
		return paths.Paths{}, fmt.Errorf("папка настроек ОС: %w", err)
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return paths.Paths{}, fmt.Errorf("домашняя папка: %w", err)
	}
	return paths.Resolve(cfg, paths.DocumentsDir(runtime.GOOS, home, os.ReadFile)), nil
}

func closeLogged(log *slog.Logger, what string, closeFn func() error) {
	if err := closeFn(); err != nil {
		log.Error("не удалось закрыть "+what, "err", err)
	}
}

package main

import (
	"context"
	"log/slog"
	"time"

	"github.com/jmoiron/sqlx"

	"pylesos/internal/api"
	"pylesos/internal/events"
	jobsrepo "pylesos/internal/jobs/repository"
	jobsservice "pylesos/internal/jobs/service"
	"pylesos/internal/paths"
	"pylesos/internal/screens"
	settingsdomain "pylesos/internal/settings/domain"
	settingsrepo "pylesos/internal/settings/repository"
	settingsservice "pylesos/internal/settings/service"
	"pylesos/web"
)

type app struct {
	server *api.Server
	runner *jobsservice.Runner
}

// wire собирает зависимости. Исполнители заданий (Worker) добавляются начиная с M2.
func wire(ctx context.Context, db *sqlx.DB, p paths.Paths, log *slog.Logger) (app, error) {
	bus := events.NewBus()
	defaults := settingsdomain.Settings{ExportsDir: p.ExportsDir, PhonePauseSec: settingsdomain.DefaultPhonePauseSec}
	settings := settingsservice.NewService(settingsrepo.NewSQLite(db), defaults)
	jobs := jobsrepo.NewSQLite(db)
	runner := jobsservice.NewRunner(ctx, jobs, bus, log, time.Now)
	if err := runner.ResumeInterrupted(ctx); err != nil {
		return app{}, err
	}
	static, err := web.Dist()
	if err != nil {
		return app{}, err
	}
	builder := screens.NewBuilder(jobs, runner, settings, time.Local)
	server, err := api.New(api.Deps{Static: static, Bus: bus, Screens: builder, Jobs: runner, Log: log})
	if err != nil {
		return app{}, err
	}
	return app{server: server, runner: runner}, nil
}

// Файл: cmd/pylesos/serve.go
package main

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"net/http"
	"path/filepath"
	"time"

	"pylesos/internal/api"
	"pylesos/internal/instance"
	"pylesos/internal/paths"
	"pylesos/internal/window"
)

// serve запускает HTTP-сервер и окно, ждёт выхода и останавливает всё по порядку:
// задания (курсоры уже сохранены) → HTTP-сервер.
func serve(ctx context.Context, cancelApp context.CancelFunc, a app, p paths.Paths, instanceFile string, log *slog.Logger) error {
	ln, err := a.server.Listen()
	if err != nil {
		return err
	}
	httpSrv := &http.Server{Handler: a.server.Handler(), ReadHeaderTimeout: 10 * time.Second}
	serveErr := make(chan error, 1)
	go func() { serveErr <- httpSrv.Serve(ln) }()
	if err := instance.Write(instanceFile, a.server.URL()); err != nil {
		log.Warn("файл экземпляра не записан — повторный запуск откроет вторую копию", "err", err)
	}
	opener := window.System(filepath.Join(p.DataDir, "window"), log)
	waitErr := waitForExit(ctx, a.server, opener, serveErr, log)
	cancelApp()
	a.runner.Wait()
	shutdownCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
	defer cancel()
	return errors.Join(waitErr, httpSrv.Shutdown(shutdownCtx))
}

// waitForExit показывает окно и ждёт выхода: закрытие окна, кнопка «Выход», Ctrl+C или сбой сервера.
func waitForExit(ctx context.Context, s *api.Server, o window.Opener, serveErr <-chan error, log *slog.Logger) error {
	closed := openWindow(s.URL(), o, log)
	for {
		select {
		case <-ctx.Done():
			return nil
		case <-s.Quit():
			return nil
		case err := <-serveErr:
			return fmt.Errorf("HTTP-сервер: %w", err)
		case <-s.Show():
			closed = openWindow(s.URL(), o, log)
		case <-closed:
			return nil
		}
	}
}

// openWindow открывает окно программы. Если окно не открылось или сразу закрылось (нет Chromium),
// страница открывается в браузере по умолчанию и возвращается nil-канал: выход — только кнопкой «Выход».
func openWindow(link string, o window.Opener, log *slog.Logger) <-chan struct{} {
	started := time.Now()
	done, err := o.OpenApp(link)
	if err != nil {
		log.Warn("окно программы не открылось — открываю в браузере по умолчанию", "err", err)
		openInBrowser(link, o, log)
		return nil
	}
	out := make(chan struct{})
	go func() {
		<-done
		if window.LivedLongEnough(time.Since(started)) {
			close(out)
			return
		}
		log.Warn("окно программы закрылось сразу — открываю в браузере по умолчанию")
		openInBrowser(link, o, log)
	}()
	return out
}

func openInBrowser(link string, o window.Opener, log *slog.Logger) {
	if err := o.OpenInOS(link, false); err != nil {
		log.Error("не удалось открыть браузер", "err", err)
	}
}
